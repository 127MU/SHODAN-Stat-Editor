cmake_minimum_required(VERSION 3.22)
project(ShodanDesktop LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(EUI_BUILD_APPS OFF CACHE BOOL "" FORCE)
set(EUI_BUILD_USER_APPS OFF CACHE BOOL "" FORCE)
set(EUI_BUILD_TEST_FIXTURES OFF CACHE BOOL "" FORCE)
set(EUI_ENABLE_MODULES OFF CACHE BOOL "" FORCE)
set(EUI_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(EUI_ENABLE_TRAY OFF CACHE BOOL "" FORCE)
set(EUI_DEPS_MODE bundled CACHE STRING "" FORCE)
include(FetchContent)
FetchContent_Declare(eui GIT_REPOSITORY https://github.com/sudoevolve/EUI-NEO.git
 GIT_TAG 1647709068e36834ab26da5a97ef4ad98aa34c9d)
FetchContent_MakeAvailable(eui)
add_executable(shodan_desktop app.cpp bridge.cpp mod_zip.cpp patch_mods.cpp mcp_server.cpp startup.cpp)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/shodan_icon.rc"
 "1 ICON \"${CMAKE_CURRENT_SOURCE_DIR}/../../offline/launcher/sseo.ico\"\n1 24 \"${CMAKE_CURRENT_SOURCE_DIR}/application.manifest\"\n")
set_property(GLOBAL PROPERTY EUI_NEO_APP_ICON_RESOURCE "${CMAKE_CURRENT_BINARY_DIR}/shodan_icon.rc")
eui_neo_configure_app(shodan_desktop)
set(LUAJIT_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/lua" CACHE PATH "LuaJIT public headers")
target_include_directories(shodan_desktop PRIVATE "${LUAJIT_INCLUDE_DIR}")
if(MSVC)
 set(LUA_IMPORT "${CMAKE_CURRENT_BINARY_DIR}/lua51.lib")
 execute_process(COMMAND "${CMAKE_AR}" /nologo /machine:x64
  "/def:${CMAKE_CURRENT_SOURCE_DIR}/lua51.def" "/out:${LUA_IMPORT}" COMMAND_ERROR_IS_FATAL ANY)
else()
 get_filename_component(COMPILER_BIN "${CMAKE_CXX_COMPILER}" DIRECTORY)
 find_program(LUA_DLLTOOL dlltool HINTS "${COMPILER_BIN}" REQUIRED)
 set(LUA_IMPORT "${CMAKE_CURRENT_BINARY_DIR}/liblua51.a")
 execute_process(COMMAND "${LUA_DLLTOOL}" -d "${CMAKE_CURRENT_SOURCE_DIR}/lua51.def"
  -l "${LUA_IMPORT}" COMMAND_ERROR_IS_FATAL ANY)
endif()
target_link_libraries(shodan_desktop PRIVATE "${LUA_IMPORT}" comdlg32 eui_zlib ws2_32 bcrypt)
target_compile_definitions(shodan_desktop PRIVATE EUI_APP_RUNNER_LIBRARY)
target_include_directories(shodan_desktop PRIVATE "${eui_SOURCE_DIR}/3rd/yyjson-0.12.0/src")
target_include_directories(shodan_desktop PRIVATE "${eui_SOURCE_DIR}/3rd/zlib-1.3.1")
if(MSVC)
 target_compile_options(shodan_desktop PRIVATE /utf-8)
else()
 target_compile_options(shodan_desktop PRIVATE -fexceptions)
endif()
set_target_properties(shodan_desktop PROPERTIES OUTPUT_NAME "SHODAN Stat Editor")
